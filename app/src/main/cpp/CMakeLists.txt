cmake_minimum_required(VERSION 3.22.1)
project(matepet)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
# Pinned for reproducible builds. To upgrade, change the tag and re-test the JNI calls in native-lib.cpp.
FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b10502 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(llama)
add_library(matepet SHARED native-lib.cpp)
target_link_libraries(matepet llama log)
